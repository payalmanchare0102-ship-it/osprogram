
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

int main()
{
    int pipe1[2], pipe2[2];
    pid_t pid;
    char pathname[100];
    char buffer[1024];
    int n;

    // Create two pipes
    if(pipe(pipe1) == -1 || pipe(pipe2) == -1)
    {
        perror("pipe");
        exit(1);
    }

    // Create child process
    pid = fork();

    if(pid < 0)
    {
        perror("fork");
        exit(1);
    }

    // Parent Process
    if(pid > 0)
    {
        // Close unused ends
        close(pipe1[0]);
        close(pipe2[1]);

        printf("Enter the pathname of the file: ");
        scanf("%99s", pathname);

        // Write pathname to first pipe
        write(pipe1[1], pathname, strlen(pathname) + 1);
        close(pipe1[1]);

        // Read file contents from second pipe
        printf("\nContents of the file:\n");

        while((n = read(pipe2[0], buffer, sizeof(buffer))) > 0)
        {
            write(STDOUT_FILENO, buffer, n);
        }

        close(pipe2[0]);

        wait(NULL);
    }

    // Child Process
    else
    {
        FILE *fp;

        // Close unused ends
        close(pipe1[1]);
        close(pipe2[0]);

        // Read pathname from first pipe
        read(pipe1[0], pathname, sizeof(pathname));
        close(pipe1[0]);

        // Open the file
        fp = fopen(pathname, "r");

        if(fp == NULL)
        {
            char *msg = "Error: File cannot be opened.\n";
            write(pipe2[1], msg, strlen(msg));
            close(pipe2[1]);
            exit(1);
        }

        // Read file and send contents
        while(fgets(buffer, sizeof(buffer), fp) != NULL)
        {
            write(pipe2[1], buffer, strlen(buffer));
        }

        fclose(fp);
        close(pipe2[1]);

        exit(0);
    }

    return 0;
}
